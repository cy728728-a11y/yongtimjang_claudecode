# 국기 원정대 지도 퀴즈 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 국기 원정대 아티팩트에 "지도 퀴즈" 모드(단계식 찾기·바로 찍기·색칠 맞히기 + 숙련도·단계·내 세계지도)를 추가하고 용팀장 계정으로 새로 발행한다.

**Architecture:** 단일 HTML 아티팩트 + 보조 파일(js/*.js, data/world.json, flags/*.svg). 순수 로직(데이터·숙련도·출제·저장)은 브라우저 전역/Node CommonJS 겸용 파일로 분리해 `node --test` 로 검증하고, 지도 렌더링(d3-geo)과 UI는 Playwright로 로컬에서 확인한다. 진행 기록은 아티팩트 `db` capability, 없으면 localStorage.

**Tech Stack:** ES5 스타일 JS(기존 코드와 동일), d3@7.9.0 + topojson-client@3.1.0 (cdn.jsdelivr.net/npm), Natural Earth 50m admin-0, mapshaper(빌드 전용), Node 24 `node:test`.

**Spec:** `docs/superpowers/specs/2026-09-24-map-quiz-design.md`

## Global Constraints

- 작업 폴더: `10-projects/국기-원정대/` (아래 경로는 모두 이 폴더 기준)
- 국기 퀴즈 기존 동작 변경 금지 (데이터를 파일로 옮기는 것만 허용)
- 한 문제 최대 10점: 단계식 대륙 3 + 나라 7, 바로 찍기 10, 색칠 맞히기 10
- 숙련도 0~3, 정답 +1 / 오답 −1, 출제 가중치 = 4 − 숙련도
- 다음 단계 해금: 현재 단계 나라의 80% 이상이 숙련도 2 이상
- 바로 찍기는 그 나라 대륙이 2단계 이상 열렸을 때만
- 러시아·튀르키예 = 유럽 (기존 게임 대륙 값 우선)
- 작은 나라 터치 영역: 화면 기준 지름 44px
- 외부 스크립트는 cdn.jsdelivr.net/npm 만, 폰 폭 375px에서 가로 스크롤 없음
- db 문서: `progress/<playerKey>` = `{ m: {iso: 0..3}, updatedAt }`
- 발행: 용팀장 계정 새 아티팩트, `capabilities: {db: {}}`, 제목 "국기 원정대"
- 코드 주석은 한국어

```js
const test = require("node:test");
const assert = require("node:assert");
const fs = require("node:fs");
const path = require("node:path");
const D = require("../js/countries.js");

const ROOT = path.join(__dirname, "..");

test("대륙별 나라 수가 CONTINENTS.count 와 같다", () => {
  assert.strictEqual(D.COUNTRIES.length, 105);
  D.CONTINENTS.forEach((c) => {
    const n = D.COUNTRIES.filter((x) => x.continent === c.key).length;
    assert.strictEqual(n, c.count, c.key);
  });
});

test("모든 나라에 iso·tier·hint 가 있다", () => {
  const seen = new Set();
  D.COUNTRIES.forEach((c) => {
    assert.match(c.iso, /^[a-z]{2}$/, c.name);
    assert.ok(!seen.has(c.iso), "중복 " + c.iso);
    seen.add(c.iso);
    assert.ok([1, 2, 3].includes(c.tier), c.name + " tier");
    assert.ok(c.hint && c.hint.length <= 45, c.name + " hint");
  });
});

test("대륙마다 1·2·3단계가 모두 있다", () => {
  D.CONTINENTS.forEach((ct) => {
    [1, 2, 3].forEach((t) => {
      assert.ok(D.COUNTRIES.some((c) => c.continent === ct.key && c.tier === t), ct.key + " " + t);
    });
  });
});

test("국기 파일이 전부 있다", () => {
  D.COUNTRIES.forEach((c) => {
    assert.ok(fs.existsSync(path.join(ROOT, "flags", c.iso + ".svg")), c.iso);
  });
});
```

```js
// Natural Earth 50m 나라 경계 → data/world.json (TopoJSON) 생성
import { existsSync, mkdirSync, readFileSync, writeFileSync, statSync } from "node:fs";
import mapshaper from "mapshaper";

const SRC_URL = "https://raw.githubusercontent.com/nvkelso/natural-earth-vector/master/geojson/ne_50m_admin_0_countries.geojson";
const RAW = "tools/.cache/ne50.geojson";
const SLIM = "tools/.cache/slim.geojson";
const OUT = "data/world.json";
// Natural Earth 대륙명 → 게임 대륙 키
const CONT = {
  "Europe": "europe", "Africa": "africa", "Asia": "asia", "Oceania": "oceania",
  "North America": "namerica", "South America": "samerica"
};

try {
  mkdirSync("tools/.cache", { recursive: true });
  mkdirSync("data", { recursive: true });

  // 1) 원본 다운로드 (캐시 있으면 생략)
  if (!existsSync(RAW)) {
    const res = await fetch(SRC_URL);
    if (!res.ok) throw new Error("다운로드 실패: HTTP " + res.status);
    writeFileSync(RAW, Buffer.from(await res.arrayBuffer()));
  }

  // 2) 속성 정리: 남극 제외, iso·cont·ko 만 남김
  const src = JSON.parse(readFileSync(RAW, "utf8"));
  const features = src.features
    .filter((f) => f.properties.CONTINENT !== "Antarctica")
    .map((f) => {
      const p = f.properties;
      const iso = p.ISO_A2_EH && p.ISO_A2_EH !== "-99" ? p.ISO_A2_EH.toLowerCase() : "";
      return {
        type: "Feature",
        geometry: f.geometry,
        properties: { iso: iso, cont: CONT[p.CONTINENT] || "", ko: p.NAME_KO || p.NAME || "" }
      };
    });
  writeFileSync(SLIM, JSON.stringify({ type: "FeatureCollection", features: features }));

  // 3) 단순화 + TopoJSON 변환
  await mapshaper.runCommands(
    `-i ${SLIM} name=countries -simplify 20% keep-shapes -o ${OUT} format=topojson quantization=100000 force`
  );
  console.log("완료:", OUT, Math.round(statSync(OUT).size / 1024) + "KB,", features.length + "개 나라");
} catch (err) {
  console.error("지도 빌드 실패:", err.message);
  process.exit(1);
}
```

```js
/* 지도 퀴즈 숙련도 계산 (순수 함수) — m = {iso: 0..3} */
(function (root) {
  "use strict";
  var MAX = 3;

  // 나라 숙련도 (없으면 0)
  function level(m, iso) {
    var v = m && m[iso];
    return typeof v === "number" ? Math.max(0, Math.min(MAX, v)) : 0;
  }

  // 정답 +1, 오답 -1 반영한 새 객체
  function applyResult(m, iso, correct) {
    var next = Object.assign({}, m || {});
    next[iso] = Math.max(0, Math.min(MAX, level(m, iso) + (correct ? 1 : -1)));
    return next;
  }

  // 출제 가중치: 모르는 나라일수록 자주
  function weight(m, iso) {
    return MAX + 1 - level(m, iso);
  }

  // 대륙의 열린 단계: 현재 단계 나라 80% 이상이 숙련도 2+ 면 다음 단계
  function unlockedTier(m, tierLists) {
    var t = 1;
    while (t < 3) {
      var list = tierLists[t] || [];
      var ok = list.filter(function (iso) { return level(m, iso) >= 2; }).length;
      if (list.length === 0 || ok / list.length >= 0.8) t++;
      else break;
    }
    return t;
  }

  // 가중치 비례 랜덤 선택
  function weightedPick(items, weightFn, rng) {
    var total = 0;
    var ws = items.map(function (it) { var w = weightFn(it); total += w; return w; });
    var r = (rng || Math.random)() * total;
    for (var i = 0; i < items.length; i++) {
      r -= ws[i];
      if (r < 0) return items[i];
    }
    return items[items.length - 1];
  }

  var api = { MAX: MAX, level: level, applyResult: applyResult, weight: weight,
              unlockedTier: unlockedTier, weightedPick: weightedPick };
  if (typeof module !== "undefined" && module.exports) module.exports = api;
  else root.MQProgress = api;
})(this);
```

```js
/* 지도 퀴즈 출제 로직 (순수 함수) */
(function (root) {
  "use strict";
  var P = (typeof module !== "undefined" && module.exports) ? require("./progress.js") : root.MQProgress;

  var POINTS = { stageContinent: 3, stageCountry: 7, direct: 10, name: 10 };

  function shuffle(arr, rng) {
    var a = arr.slice();
    for (var i = a.length - 1; i > 0; i--) {
      var j = Math.floor(rng() * (i + 1));
      var t = a[i]; a[i] = a[j]; a[j] = t;
    }
    return a;
  }

  // 대륙의 단계별 나라 목록
  function tierLists(countries, contKey) {
    var t = { 1: [], 2: [], 3: [] };
    countries.forEach(function (c) { if (c.continent === contKey) t[c.tier].push(c.iso); });
    return t;
  }

  function unlockedByContinent(countries, m, contKeys) {
    var out = {};
    contKeys.forEach(function (k) { out[k] = P.unlockedTier(m, tierLists(countries, k)); });
    return out;
  }

  // 출제 풀 = 선택 대륙 중 열린 단계 이하 나라
  function poolFor(countries, m, contKeys) {
    var unlocked = unlockedByContinent(countries, m, contKeys);
    var pool = countries.filter(function (c) {
      return contKeys.indexOf(c.continent) > -1 && c.tier <= unlocked[c.continent];
    });
    return { pool: pool, unlocked: unlocked };
  }

  // 색칠 맞히기 보기 5개 (같은 대륙 나라 우선)
  function nameOptions(target, countries, rng) {
    rng = rng || Math.random;
    var same = shuffle(countries.filter(function (c) {
      return c.continent === target.continent && c.iso !== target.iso;
    }), rng);
    var others = shuffle(countries.filter(function (c) { return c.continent !== target.continent; }), rng);
    var picks = same.concat(others).slice(0, 4).map(function (c) { return c.name; });
    return shuffle([target.name].concat(picks), rng);
  }

  // 문제 하나 생성
  function makeQuestion(opts) {
    var rng = opts.rng || Math.random;
    var recent = opts.recent || [];
    var r = poolFor(opts.countries, opts.m, opts.contKeys);
    var fresh = r.pool.filter(function (c) { return recent.indexOf(c.iso) === -1; });
    var candidates = fresh.length ? fresh : r.pool;
    var target = P.weightedPick(candidates, function (c) { return P.weight(opts.m, c.iso); }, rng);

    var types = ["stage", "name"];
    if (r.unlocked[target.continent] >= 2) types.push("direct");
    var type = types[Math.floor(rng() * types.length)];

    var q = { country: target, type: type };
    if (type === "name") q.options = nameOptions(target, opts.countries, rng);
    return q;
  }

  var api = { POINTS: POINTS, tierLists: tierLists, unlockedByContinent: unlockedByContinent,
              poolFor: poolFor, nameOptions: nameOptions, makeQuestion: makeQuestion };
  if (typeof module !== "undefined" && module.exports) module.exports = api;
  else root.MQLogic = api;
})(this);
```

```js
/* 지도 퀴즈 진행 기록 저장소: 아티팩트 db (여러 기기 공유) + localStorage 사본 */
(function (root) {
  "use strict";
  var LS_KEY = "mq-progress-v1";

  function createStore(opts) {
    var db = opts.db || null;
    var storage = opts.storage || null;
    var cache = {};
    var chains = {};

    function lsRead() {
      if (!storage) return {};
      try { return JSON.parse(storage.getItem(LS_KEY)) || {}; } catch (e) { return {}; }
    }
    function lsWrite() {
      if (!storage) return;
      try { storage.setItem(LS_KEY, JSON.stringify(cache)); } catch (e) { /* 저장 공간 없음 → 무시 */ }
    }

    return {
      mode: db ? "db" : "local",

      // 플레이어 기록 불러오기 (db 우선, 실패하면 localStorage 사본)
      load: function (keys) {
        var local = lsRead();
        if (!db) {
          keys.forEach(function (k) { cache[k] = local[k] || {}; });
          return Promise.resolve();
        }
        return Promise.all(keys.map(function (k) {
          return db.doc("progress/" + k).get().then(function (snap) {
            var body = snap.exists ? snap.data() : null;
            cache[k] = body && body.m ? Object.assign({}, body.m) : (local[k] || {});
          }).catch(function (e) {
            console.warn("[MQStore] 불러오기 실패", k, e && e.code);
            cache[k] = local[k] || {};
          });
        }));
      },

      get: function (key) { return cache[key] || {}; },

      // 저장: 캐시·localStorage 즉시, db 는 문서별로 하나씩
      save: function (key, m) {
        cache[key] = m;
        lsWrite();
        if (!db) return Promise.resolve();
        var prev = chains[key] || Promise.resolve();
        chains[key] = prev.then(function () {
          return db.doc("progress/" + key).set({ m: m, updatedAt: Date.now() });
        }).catch(function (e) {
          console.warn("[MQStore] 저장 실패", key, e && e.code);
        });
        return chains[key];
      }
    };
  }

  // 브라우저용: db 가능하면 db, 아니면 localStorage 만
  function connect() {
    var storage = null;
    try { storage = root.localStorage; storage.getItem(LS_KEY); } catch (e) { storage = null; }
    var use = root.claude && typeof root.claude.use === "function"
      ? root.claude.use("db") : Promise.resolve(null);
    return Promise.resolve(use).catch(function () { return null; }).then(function (db) {
      return createStore({ db: db, storage: storage });
    });
  }

  var api = { createStore: createStore, connect: connect, LS_KEY: LS_KEY };
  if (typeof module !== "undefined" && module.exports) module.exports = api;
  else root.MQStore = api;
})(this);
```

```js
/* 지도 퀴즈 세계지도 렌더링 (d3-geo + topojson) */
(function (root) {
  "use strict";
  var W = 800, H = 500;
  var HIT_PX = 22;   // 작은 나라 터치 반지름 (화면 px)
  var PIN_PX = 5;    // 작은 나라 점 반지름 (화면 px)
  var LABEL_PX = 14; // 이름표 글자 크기 (화면 px)
  var SMALL_PX = 14; // 이보다 작게 보이는 나라는 점 추가

  // 대륙별 확대 범위 [[서경, 남위], [동경, 북위]]
  var VIEWS = {
    europe:   { bounds: [[-25, 34], [45, 71]] },
    africa:   { bounds: [[-20, -36], [52, 38]] },
    asia:     { bounds: [[25, -11], [150, 55]] },
    oceania:  { bounds: [[110, -48], [-168, 2]], rotate: [-150, 0] },
    namerica: { bounds: [[-168, 7], [-52, 72]] },
    samerica: { bounds: [[-82, -56], [-34, 13]] }
  };

  // 경계 상자를 점 묶음으로 (fitExtent 용)
  function boxGeo(b) {
    var w = b[0][0], s = b[0][1], e = b[1][0], n = b[1][1];
    if (e < w) e += 360;
    var pts = [];
    for (var i = 0; i <= 4; i++) {
      var lon = w + (e - w) * i / 4;
      if (lon > 180) lon -= 360;
      pts.push([lon, s], [lon, n]);
    }
    return { type: "MultiPoint", coordinates: pts };
  }

  function create(svgEl, topo, opts) {
    var obj = topo.objects.countries;
    var features = root.topojson.feature(topo, obj).features;
    var nb = root.topojson.neighbors(obj.geometries);
    var byIso = {};
    features.forEach(function (f, i) {
      f._i = i;
      if (f.properties.iso) byIso[f.properties.iso] = f;
    });

    var svg = root.d3.select(svgEl).attr("viewBox", "0 0 " + W + " " + H);
    svg.selectAll("*").remove();
    var layer = svg.append("g").attr("class", "mq-layer");
    var gLand = layer.append("g");
    var gDots = layer.append("g");
    var gLabels = layer.append("g");
    var tap = null, path = null, k = 1;

    // viewBox 1단위가 화면에서 몇 px 인지
    function screenScale() {
      var w = svgEl.getBoundingClientRect().width;
      return w > 0 ? w / W : 1;
    }
    function px(v) { return v / (screenScale() * k); }

    function resize() {
      gDots.selectAll("circle.mq-hit").attr("r", px(HIT_PX));
      gDots.selectAll("circle.mq-pin").attr("r", px(PIN_PX));
      gLabels.selectAll("text").style("font-size", px(LABEL_PX) + "px").style("stroke-width", px(3) + "px");
    }

    var zoom = root.d3.zoom()
      .scaleExtent([1, 8])
      .translateExtent([[0, 0], [W, H]])
      .on("zoom", function (ev) {
        k = ev.transform.k;
        layer.attr("transform", ev.transform);
        resize();
      });
    svg.call(zoom);

    function fire(ev, f) { if (tap) tap(f, ev); }

    function projectionFor(key) {
      var p;
      if (key === "world") {
        p = root.d3.geoNaturalEarth1().fitExtent([[8, 8], [W - 8, H - 8]], { type: "Sphere" });
      } else {
        var v = VIEWS[key];
        p = root.d3.geoMercator();
        if (v.rotate) p.rotate(v.rotate);
        p.fitExtent([[12, 12], [W - 12, H - 12]], boxGeo(v.bounds));
      }
      return p;
    }

    function show(key) {
      path = root.d3.geoPath(projectionFor(key));
      k = 1;
      svg.call(zoom.transform, root.d3.zoomIdentity);
      layer.attr("transform", null);
      clearAll();

      gLand.selectAll("path")
        .data(features, function (f) { return f._i; })
        .join("path")
        .attr("d", path)
        .attr("class", function (f) { return "mq-land" + (opts.isQuiz(f.properties.iso) ? " mq-quiz" : ""); })
        .on("click", fire);

      // 화면에서 너무 작은 문제 나라는 점 + 넓은 터치 영역
      var s = screenScale();
      var small = features.filter(function (f) {
        if (!opts.isQuiz(f.properties.iso)) return false;
        var b = path.bounds(f);
        if (!isFinite(b[0][0])) return false;
        var c = path.centroid(f);
        if (!(c[0] >= 0 && c[0] <= W && c[1] >= 0 && c[1] <= H)) return false;
        return (b[1][0] - b[0][0]) * s < SMALL_PX && (b[1][1] - b[0][1]) * s < SMALL_PX;
      });
      var dots = gDots.selectAll("g.mq-dot").data(small, function (f) { return f._i; });
      dots.exit().remove();
      var enter = dots.enter().append("g").attr("class", "mq-dot");
      enter.append("circle").attr("class", "mq-hit");
      enter.append("circle").attr("class", "mq-pin");
      gDots.selectAll("g.mq-dot")
        .attr("transform", function (f) { return "translate(" + path.centroid(f) + ")"; })
        .on("click", fire);
      resize();
    }

    // iso 에 해당하는 나라 모양 + 점
    function nodesFor(iso) {
      var match = function (f) { return f.properties.iso === iso; };
      return gLand.selectAll("path").filter(match).nodes()
        .concat(gDots.selectAll("g.mq-dot").filter(match).nodes());
    }

    function mark(iso, cls) {
      if (!iso) return;
      nodesFor(iso).forEach(function (n) { n.classList.add(cls); });
    }

    function markContinent(key) {
      gLand.selectAll("path").filter(function (f) { return opts.contOf(f) === key; })
        .classed("mq-cont", true);
    }

    function label(iso, text, cls) {
      var f = byIso[iso];
      if (!f || !path) return;
      var c = path.centroid(f);
      if (!isFinite(c[0])) return;
      gLabels.append("text")
        .attr("class", "mq-label " + (cls || ""))
        .attr("x", c[0]).attr("y", c[1])
        .attr("dy", "-0.6em")
        .text(text);
      resize();
    }

    function clearLabels() { gLabels.selectAll("*").remove(); }

    function clearAll() {
      svg.selectAll(".mq-target,.mq-correct,.mq-wrong,.mq-cont,.mq-m1,.mq-m2,.mq-m3")
        .classed("mq-target mq-correct mq-wrong mq-cont mq-m1 mq-m2 mq-m3", false);
      clearLabels();
    }

    function setMastery(m) {
      Object.keys(m || {}).forEach(function (iso) {
        var lv = m[iso];
        if (lv >= 1 && lv <= 3 && opts.isQuiz(iso)) mark(iso, "mq-m" + lv);
      });
    }

    // 이웃 나라 (문제 나라 먼저)
    function neighborsOf(iso) {
      var f = byIso[iso];
      if (!f) return [];
      var list = nb[f._i].map(function (i) { return features[i].properties.iso; })
        .filter(function (x) { return x && x !== iso; });
      return list.filter(opts.isQuiz).concat(list.filter(function (x) { return !opts.isQuiz(x); }));
    }

    function nameOfIso(iso) { return byIso[iso] ? opts.nameOf(byIso[iso]) : iso; }

    return {
      show: show,
      onTap: function (fn) { tap = fn; },
      mark: mark,
      markContinent: markContinent,
      label: label,
      clearLabels: clearLabels,
      setMastery: setMastery,
      neighborsOf: neighborsOf,
      contOf: opts.contOf,
      nameOf: opts.nameOf,
      nameOfIso: nameOfIso
    };
  }

  root.MQMapView = { create: create, VIEWS: VIEWS };
})(this);
```

```js
(function (root) {
  "use strict";
  var D = root.MQData, L = root.MQLogic, P = root.MQProgress;
  var CONT_NAMES = {};
  D.CONTINENTS.forEach(function (c) { CONT_NAMES[c.key] = c.name; });
  var byIso = {};
  D.COUNTRIES.forEach(function (c) { byIso[c.iso] = c; });

  var topo = null, store = null, readyP = null, recent = [];

  // 받침 있으면 a, 없으면 b (예: 영국은 / 프랑스는)
  function josa(word, a, b) {
    var code = word.charCodeAt(word.length - 1) - 0xAC00;
    if (code < 0 || code > 11171) return b;
    return code % 28 ? a : b;
  }

  // 지도 파일 + 저장소 준비 (한 번만)
  function ready(playerKeys) {
    if (!readyP) {
      readyP = Promise.all([
        fetch("data/world.json").then(function (r) {
          if (!r.ok) throw new Error("지도 파일을 못 불러왔어요 (" + r.status + ")");
          return r.json();
        }),
        root.MQStore.connect()
      ]).then(function (res) {
        topo = res[0];
        store = res[1];
        return store.load(playerKeys);
      }).catch(function (e) {
        readyP = null;
        throw e;
      });
    }
    return readyP;
  }

  function makeView(svgEl) {
    return root.MQMapView.create(svgEl, topo, {
      isQuiz: function (iso) { return !!byIso[iso]; },
      contOf: function (f) {
        var c = byIso[f.properties.iso];
        return (c ? c.continent : f.properties.cont) || null;
      },
      nameOf: function (f) {
        var c = byIso[f.properties.iso];
        return c ? c.name : (f.properties.ko || "이름 없는 땅");
      }
    });
  }

  function renderQuestion(card, ctx) {
    var q = L.makeQuestion({ countries: D.COUNTRIES, m: store.get(ctx.playerKey), contKeys: ctx.contKeys, recent: recent });
    recent.push(q.country.iso);
    if (recent.length > 6) recent.shift();
    var c = q.country;
    var tag = { stage: "지도에서 찾기", direct: "바로 찍기", name: "색칠된 나라" }[q.type];

    card.innerHTML =
      '<div class="mode-tag">' + tag + '</div>' +
      '<div class="turn-banner"><span class="picon">' + ctx.player.icon + '</span>' + ctx.player.name + '의 차례!</div>' +
      '<div class="mq-prompt" id="mqPrompt"></div>' +
      '<div class="mq-map"><svg id="mqSvg" role="img" aria-label="세계지도"></svg></div>' +
      '<div class="choices" id="choices"></div>' +
      '<div class="feedback" id="feedback"></div>' +
      '<div class="mq-review" id="mqReview" hidden></div>' +
      '<div class="score-board" id="scoreBoard"></div>';

    var view = makeView(card.querySelector("#mqSvg"));
    var promptEl = card.querySelector("#mqPrompt");
    var fb = card.querySelector("#feedback");
    var flagImg = '<img class="mq-flag" src="flags/' + c.iso + '.svg" alt="국기">';
    var nameEul = c.name + josa(c.name, "을", "를");
    var nameEun = c.name + josa(c.name, "은", "는");
    var points = 0, done = false;

    // 문제 끝: 점수·숙련도 반영 + 복습 표시
    function result(correct, userAnswer) {
      if (done) return;
      done = true;
      view.onTap(null);
      store.save(ctx.playerKey, P.applyResult(store.get(ctx.playerKey), c.iso, correct));

      fb.textContent = correct
        ? ctx.player.name + " 정답이에요! (+" + points + "pt)"
        : (points ? "대륙은 맞았어요! (+" + points + "pt) " : "아쉬워요. ") + "정답은 " + c.name + josa(c.name, "이에요", "예요") + ".";

      view.mark(c.iso, correct ? "mq-correct" : "mq-target");
      var nbs = view.neighborsOf(c.iso).slice(0, 3);
      nbs.forEach(function (iso) { view.label(iso, view.nameOfIso(iso), "mq-label-soft"); });
      view.label(c.iso, c.name, "mq-label-main");

      var rv = card.querySelector("#mqReview");
      rv.innerHTML =
        '<div class="mq-review-title">📍 ' + c.name + ' · ' + CONT_NAMES[c.continent] + '</div>' +
        '<div>' + c.hint + '</div>' +
        (nbs.length ? '<div class="mq-nb">이웃 나라: ' + nbs.map(view.nameOfIso).join(" · ") + '</div>' : '');
      rv.hidden = false;

      ctx.onDone({ points: points, correct: correct, userAnswer: userAnswer, correctAnswer: c.name, country: c });

    // 틀리게 누른 나라 표시
    function showWrongTap(f) {
      var iso = f.properties.iso;
      view.mark(iso, "mq-wrong");
      view.label(iso, "여긴 " + view.nameOf(f), "mq-label-wrong");
    }

      });
    } else if (q.type === "direct") {
      promptEl.innerHTML = flagImg + '<span>세계지도에서 <b>' + nameEul + '</b> 바로 찾아보세요!<small>두 손가락(또는 마우스 휠)으로 확대할 수 있어요</small></span>';
      view.show("world");
      view.onTap(function (f) {
        var hit = f.properties.iso === c.iso;
        if (hit) points += L.POINTS.direct;
        else showWrongTap(f);
        result(hit, hit ? c.name : view.nameOf(f));
      });
    } else {
      promptEl.innerHTML = '<span>반짝이는 나라는 어디일까요?</span>';
      view.show(c.continent);
      view.mark(c.iso, "mq-target");
      var choicesEl = card.querySelector("#choices");
      q.options.forEach(function (opt, i) {
        var b = document.createElement("button");
        b.className = "choice";
        b.innerHTML = '<span class="letter">' + "ABCDE"[i] + '</span><span>' + opt + '</span>';
        b.addEventListener("click", function () {
          if (done) return;
          var hit = opt === c.name;
          if (hit) points += L.POINTS.name;

            x.disabled = true;

            else if (x === b) x.classList.add("wrong");

          });

        });
        choicesEl.appendChild(b);
      });
    }
  }

  // 내 세계지도: 숙련도만큼 색칠
  function renderAtlas(card, player, onBack) {

      '<div class="score-board">' + D.CONTINENTS.map(function (ct) {
        return '<span class="score-chip">' + ct.icon + ' ' + ct.name + ' <span class="pts">' + unlocked[ct.key] + '단계</span></span>';
      }).join("") + '</div>' +
      '<div class="hint">나라를 누르면 이름이 보여요</div>' +

    var view = makeView(card.querySelector("#mqSvg"));

    view.setMastery(m);
    view.onTap(function (f) {
      view.clearLabels();
      view.label(f.properties.iso, view.nameOf(f), "mq-label-main");
    });
    card.querySelector("#mqBack").addEventListener("click", onBack);
  }

  root.MQMapQuiz = {
    ready: ready,
    renderQuestion: renderQuestion,
    renderAtlas: renderAtlas,
    storeMode: function () { return store ? store.mode : null; }
  };
})(this);
```

```js
function beginGame(){
  state.turnOrder = shuffle(state.selectedPlayers);
  state.totalQ = QUESTIONS_PER_PLAYER * state.turnOrder.length;
```

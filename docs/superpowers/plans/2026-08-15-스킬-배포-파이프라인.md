```python
"""플러그인에 복사된 라이브러리가 정본과 같은지 — 어긋나면 배포를 막는다.

사본이 여러 개라 시간이 지나면 반드시 갈라진다. 사람이 기억으로 맞추지 않고
빌더가 복사하고 이 테스트가 확인한다.
"""
import hashlib
import os
import unittest

ROOT = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
SHARED = os.path.join(ROOT, "shared", "eroomlib")
PLUGINS = os.path.join(ROOT, "plugins")


def _digest(path):
    h = hashlib.sha256()
    with open(path, "rb") as f:
        h.update(f.read())
    return h.hexdigest()


def _py_files(d):
    return sorted(f for f in os.listdir(d) if f.endswith(".py"))


class TestLibSync(unittest.TestCase):
    def test_정본이_존재한다(self):
        self.assertTrue(os.path.isdir(SHARED), "shared/eroomlib 이 없다")
        self.assertIn("localstore.py", _py_files(SHARED))

    def test_모든_플러그인_사본이_정본과_같다(self):
        want = {f: _digest(os.path.join(SHARED, f)) for f in _py_files(SHARED)}
        checked = 0
        for name in sorted(os.listdir(PLUGINS)):
            lib = os.path.join(PLUGINS, name, "lib", "eroomlib")
            if not os.path.isdir(lib):
                continue
            got = {f: _digest(os.path.join(lib, f)) for f in _py_files(lib)}
            self.assertEqual(got, want,
                             "%s 의 라이브러리 사본이 정본과 다르다 — "
                             "python scripts/build_plugins.py 로 다시 빌드하라" % name)
            checked += 1
        self.assertGreater(checked, 0, "라이브러리를 가진 플러그인이 하나도 없다")


if __name__ == "__main__":
    unittest.main()
```

```python
def _resolve_sheet(args):
    """작업 대상 그룹의 진행판 파일 경로를 정한다."""
    if getattr(args, "sheet", None):
        return args.sheet
    name = (getattr(args, "group_name", "") or "").strip()
    if not name:
        raise RuntimeError("--sheet 또는 --group-name 중 하나는 필요하다")
    from eroomlib.config import store_for_group
    return store_for_group(name)


def _bootstrap_board(sheet, group_name):
    """진행판이 없으면 그룹 상품으로 만든다(있으면 그대로 — 멱등)."""
    if not group_name:
        return
    matrix.ensure(sheet)
    if matrix.read(sheet):
        return
    print(f"  진행판 최초 생성 — 마켓그룹 '{group_name}' 상품 목록을 읽는다")
    products = matrix._group_products(group_name)
    added = matrix.sync_products(sheet, products)
    print(f"  진행판에 상품 {added}건 등록")
```

`cmd_prep` 진입부에서 `_resolve_sheet` 직후 `_bootstrap_board(sheet, ...)` 를 부른다.

```python
def resolve_keyword_dir(explicit=None):
    """키워드 원본 엑셀이 있는 폴더. 인자 → 환경변수 → 홈 기본값 순."""
    if explicit:
        return os.path.abspath(os.path.expanduser(explicit))
    env = os.environ.get("BULSAJA_KIT_KEYWORDS")
    if env:
        return os.path.abspath(os.path.expanduser(env))
    return os.path.join(os.path.expanduser("~"), "bulsaja-kit", "keywords")


def require_keyword_files(d):
    """폴더 안의 키워드 엑셀 목록. 없으면 **안내하고 멈춘다.**

    조용히 0건으로 진행하면 사용자는 작업이 끝난 줄 오해한다.
    """
    files = []
    if os.path.isdir(d):
        files = sorted(os.path.join(d, f) for f in os.listdir(d)
                       if f.lower().endswith((".xlsx", ".xls")))
    if files:
        return files
    sys.stderr.write(
        "\n키워드 파일이 없습니다.\n"
        "  넣을 곳: %s\n"
        "  넣을 것: 셀러라이프에서 받은 카테고리 키워드 엑셀(.xlsx)\n"
        "  폴더가 없으면 만들어서 파일을 넣고 다시 실행하세요.\n\n" % d)
    raise SystemExit(2)
```
